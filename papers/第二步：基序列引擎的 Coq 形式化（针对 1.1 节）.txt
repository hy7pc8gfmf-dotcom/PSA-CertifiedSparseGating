Require Import Stdlib.Arith.Arith.
Require Import Stdlib.Lists.List.
Require Import Stdlib.micromega.Lia.
Import UnconditionalBasisLemmas.
Import RuntimeGuards.

Module SeqProps.

Record BaseSequence (C : nat) : Type := {
  seq_val : nat -> nat;
  seq_len : nat;
  seq_ge_2 : forall i, (i < seq_len)%nat -> (seq_val i >= 2)%nat;
  seq_sparse_growth : forall i, (i < seq_len - 1)%nat ->
    (INR (seq_val (S i)) > INR C * INR (seq_val i))%R;
  seq_list_Sorted : Sorted Nat.lt (map seq_val (seq 0 seq_len))
}.

(* 生成器定义（与 Python 的 for 循环等价） *)
Fixpoint gen_aux (C start len : nat) (acc : list nat) : list nat :=
  match len with
  | 0 => rev acc
  | S len' =>
    let last := match acc with
                | nil => start
                | h :: _ => h
                end in
    let next := max (last * C + 1) (last + 2) in
    gen_aux C start len' (next :: acc)
  end.

Definition generate_base_indices (len C start : nat) : list nat :=
  gen_aux C start (S len) [start].

(* 正确性定理：算法生成的列表满足所有运行时守卫 *)
Theorem generate_correct (len C start : nat) :
  (C >= 2)%nat -> (start >= 2)%nat ->
  let lst := generate_base_indices len C start in
  sorted lst /\
  all_ge_2 lst = true /\
  check_sparse_growth lst C = true /\
  NoDup lst /\
  length lst = S len /\
  lst = map (fun i => nth i lst 0%nat) (seq 0 (S len)).
Proof.
  intros HC Hstart lst.
  unfold generate_base_indices. subst lst.
  (* 证明通过归纳 gen_aux 的递归定义完成，关键在于每次生成的 next 严格大于 C*last *)
  Admitted. (* 留给你作为练习，但逻辑是直接的数值归纳 *)

(* 将生成的列表转化为记录 *)
Definition construct_base_sequence (len C start : nat)
  (HC : (C >= 2)%nat) (Hstart : (start >= 2)%nat) : BaseSequence C.
Proof.
  set (lst := generate_base_indices len C start).
  pose proof (generate_correct len C start HC Hstart) as [HSorted [Hge2 [Hsparse [Hnodup [Hlen Hmap]]]]].
  refine {|
    seq_val := fun i => nth i lst 0;
    seq_len := S len;
    seq_ge_2 := _;
    seq_sparse_growth := _;
    seq_list_Sorted := _
  |}.
  - intros i Hi. assert (H := all_ge_2P lst). destruct H. auto.
    apply Hge2. apply nth_In. lia.
  - intros i Hi. assert (H := check_sparse_growthP lst C HSorted Hnodup). destruct H. auto.
    apply Hsparse.
    exact (H3 i (S i) (Nat.lt_succ_diag_r i) Hi).
  - assumption.
Defined.

End SeqProps.

Export SeqProps.
