(* ---------- 1. 基值≥2的布尔判定 ---------- *)
Definition all_ge_2 (vals : list nat) : bool :=
  List.forallb (fun v => 2 <=? v) vals.

Lemma all_ge_2P (vals : list nat) :
  reflect (forall v, In v vals -> (v >= 2)%nat) (all_ge_2 vals).
Proof.
  apply (iff_reflect_forall _ (fun v => Nat.leb_spec 2 v)).
  intro v. rewrite List.forallb_forall. reflexivity.
Qed.

(* ---------- 2. 相邻元素稀疏增长的布尔判定 ---------- *)
Fixpoint forallb_adjacent (l : list nat) (f : nat -> nat -> bool) : bool :=
  match l with
  | [] | [_] => true
  | h1 :: h2 :: t => f h1 h2 && forallb_adjacent (h2 :: t) f
  end.

Definition check_sparse_growth (indices : list nat) (C : nat) : bool :=
  forallb_adjacent indices (fun a b => (C * a) <? b).

(* 关键引理：列表有序是全局增长的充分条件（解耦判定与索引） *)
Lemma forallb_adjacent_sorted_implies (l : list nat) (f : nat -> nat -> bool) :
  Sorted Nat.lt l ->
  (forall a b, f a b = true -> a < b) ->
  forallb_adjacent l f = true ->
  (forall i j, (i < j)%nat -> (j < length l)%nat ->
    let a := nth i l 0 in
    let b := nth j l 0 in
    f a b = true).
Proof.
  induction l as [|h1 l IH]; intros HSorted Hprop Hfb i j Hij Hjlen.
  - simpl in Hjlen; lia.
  - destruct l as [|h2 t].
    + simpl in Hjlen; lia.
    + simpl in Hfb; apply andb_prop in Hfb as [Hf12 Hfb_tail].
      inversion HSorted as [|? ? Hlt HSorted_tail]; subst.
      destruct i as [|i'], j as [|j']; try lia.
      * subst a b; simpl; exact Hf12.
      * simpl.
        assert (Hj'len : (j' < length t)%nat) by (simpl in Hjlen; lia).
        apply (IH _ HSorted_tail); try assumption.
        intros i'' j'' Hlt'' Hlen''.
        apply (IHf (fun x y => f x y = true -> x < y)); auto.
        intros a' b' Hf' Hlen''.
        eapply IH; eauto.
        exact Hprop.
      * exfalso; simpl in Hij; lia.
      * apply (IH t HSorted_tail Hprop Hfb_tail i' j'); auto with arith.
Qed.

Lemma check_sparse_growthP (indices : list nat) (C : nat) :
  Sorted Nat.lt indices ->
  NoDup indices ->
  reflect (forall i j, (i < j)%nat -> (j < length indices)%nat ->
    INR (nth j indices 0%nat) > INR C * INR (nth i indices 0%nat))%R
    (check_sparse_growth indices C).
Proof.
  intros HSorted HDup.
  assert (Hstrict : forall a b, (C * a <? b)%nat = true -> (a < b)%nat).
  { intros. apply Nat.ltb_lt in H. destruct (Nat.le_gt_cases a b); auto.
    exfalso. apply Nat.lt_irrefl with b. eapply Nat.lt_le_trans; [apply Nat.lt_mul_l|]; eauto. lia. }
  destruct (forallb_adjacent indices (fun a b => (C * a <? b)%nat)) eqn:Hall.
  - left.
    intros i j Hij Hjlen.
    specialize (forallb_adjacent_sorted_implies indices (fun a b => (C * a <? b)%nat) HSorted Hstrict Hall i j Hij Hjlen).
    intro Heq. apply Nat.ltb_lt in Heq.
    apply INR_lt. exact Heq.
  - right. intro Hbad.
    assert (Href := forallb_adjacent_sorted_implies indices (fun a b => (C * a <? b)%nat) HSorted Hstrict Hall).
    (* 反证，构造一对相邻矛盾 *)
    clear - Hbad Hall.
    induction indices as [|h t IH]; simpl in *; try discriminate.
    destruct t; simpl in *; try discriminate.
    destruct (Nat.ltb_spec (C * h) n).
    + apply IH in Hall; intuition.
    + exfalso. apply Hbad with (i := 0%nat) (j := 1%nat); simpl; lia.
    + exfalso. apply Hbad with (i := 0%nat) (j := 1%nat); simpl; try lia.
      apply Rlt_not_ge. apply (INR_lt 10). lia.
  - intros Hf; exact Hf.
Qed.

(* ---------- 3. c-稀疏子集验证 ---------- *)
Definition check_c_sparse_on_vals (vals : list nat) (C : nat) : bool :=
  forallb_adjacent vals (fun a b => (C * a) <? b) && all_ge_2 vals.

Lemma check_c_sparse_on_valsP (vals : list nat) (C : nat) :
  Sorted Nat.lt vals ->
  reflect (c_sparse_subset (fun i => nth i vals 0%nat) C (seq 0 (length vals)))
          (check_c_sparse_on_vals vals C).
Proof.
  intros HSorted.
  unfold c_sparse_subset; cbv beta.
  assert (Hnodup : NoDup (seq 0 (length vals))).
  { apply seq_NoDup. }
  assert (H_len : forall i, (i < length (seq 0 (length vals)))%nat <-> (i < length vals)%nat).
  { split; intro H; rewrite length_seq in *; exact H. }
  destruct (check_c_sparse_on_vals vals C) eqn:Hcheck; cbv beta in Hcheck.
  - left. split.
    + apply seq_NoDup.
    + unfold check_c_sparse_on_vals in Hcheck.
      apply andb_prop in Hcheck as [Hsparse Hge2].
      assert (Hall_ge2P := all_ge_2P vals); destruct Hall_ge2P as [Hprop_ge2 | Hprop_not].
      * clear Hprop_not.
        assert (HsparseP := check_sparse_growthP vals C HSorted Hnodup).
        destruct HsparseP as [Hsparse_prop | Hsparse_not].
        -- intros i j Hi Hj Hij.
           apply (Hsparse_prop i j Hij).
           rewrite length_seq in Hj; exact Hj.
        -- exfalso. apply Hsparse_not. exact Hsparse.
      * exfalso. apply Hprop_ge2. exact Hge2.
  - right. intro Hsparse.
    destruct Hsparse as [_ Hgrowth].
    assert (Hge2 : forall v, In v vals -> (v >= 2)%nat).
    { intros v Hin. apply Hgrowth with (i0 := 0%nat) (j := 1%nat) in Hin; simpl in Hin.
      lia. }
    assert (Hsparse_list : forall i j, (i < j)%nat -> (j < length vals)%nat ->
      INR (nth j vals 0%nat) > INR C * INR (nth i vals 0%nat))%R.
    { intros i j Hij Hjlen. exact (Hgrowth i j Hij (proj2 (H_len j) Hjlen)). }
    (* 用 growth 推出 check 为真，矛盾 *)
    clear - Hcheck Hsparse_list HSorted.
    (* 此处需要一个归纳：由全局增长推出相邻增长为真 *)
    admit. (* 暂留一个可完成的证明缺口 *)
Abort.

End RuntimeGuards.
