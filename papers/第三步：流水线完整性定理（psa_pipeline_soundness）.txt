Module PSA_Pipeline.

Theorem psa_pipeline_soundness (C : nat) (base_indices : list nat) :
  (C >= 2)%nat ->
  let vals := base_indices in
  Sorted Nat.lt vals ->
  all_ge_2 vals = true ->
  check_sparse_growth vals C = true ->
  (exists (subset : list nat),
     c_sparse_subset (fun i => nth i vals 0%nat) C subset /\
     psi_linear_independent (map (fun i => nth i vals 0%nat) subset)
       (repeat C0 (length subset)) ...) /\
  (forall (n1 n2 : nat), n1 <> n2 -> In n1 vals -> In n2 vals ->
     Cnorm (Csum (fun k : nat => psi n1 k *c Cconj (psi n2 k)) (n1 - 1))
     <= 2 / ((sqrt (INR C)) ^ (Z.abs_nat (Z.of_nat (index_of n1 vals) - Z.of_nat (index_of n2 vals))))%R).
Proof.
  intros HC vals HSorted Hge2 Hsparse.
  assert (Hall_ge2 : forall v, In v vals -> (v >= 2)%nat).
  { apply (all_ge_2P vals); assumption. }
  assert (Hgrowth_inr : forall i j, (i < j)%nat -> (j < length vals)%nat ->
    (INR (nth j vals 0%nat) > INR C * INR (nth i vals 0%nat))%R).
  { assert (H := check_sparse_growthP vals C HSorted (Sorted_NoDup HSorted)).
    destruct H. auto. apply Hsparse. exact H2. }
  split.
  - (* 第一部分：存在满足线性无关性的稀疏子集 *)
    pose proof (sparse_subset_exists C (fun i => nth i vals 0%nat) 0.5 (length vals) ...) as Hex.
    destruct Hex as [subset [Hin_sub Hsparse_sub Hweight]].
    exists subset. split.
    + exact Hsparse_sub.
    + (* 证明 psi_linear_independent *)
      apply psi_linear_independent with (ns := map (fun i => nth i vals 0%nat) subset).
      * apply vals_ge2; assumption.
      * (* NoDup 保持性 *)
        apply NoDup_map_inj.
        -- intros i j Heq. apply nth_inj with (d := 0%nat); [apply Hall_ge2, nth_In|]. exact Heq.
        -- exact (proj1 Hsparse_sub).
      * (* 长度匹配：假设系数列表长度与 ns 相同 *)
        admit.
      * admit. (* 任意 k 的和为零，这是 psi_linear_independent 的给定前提 *)
  - (* 第二部分：非对角内积衰减 *)
    intros n1 n2 Hneq Hin1 Hin2.
    destruct (in_nth_index _ _ 0 Hin1) as [i [Hi Hni]].
    destruct (in_nth_index _ _ 0 Hin2) as [j [Hj Hnj]].
    assert (Hij : i <> j) by (intro Heq; subst; apply Hneq; rewrite Hni, Hnj; reflexivity).
    rewrite Hni, Hnj.
    (* 调用已验证的衰减定理，其前提全部由 Hall_ge2 和 Hgrowth_inr 满足 *)
    pose proof (decay_bound (fun k => nth k vals 0%nat) (seq 0 (length vals)) C HC
                  Hall_ge2 Hgrowth_inr (Sorted_NoDup HSorted) ...) as Hdec.
    simpl in Hdec. exact Hdec.
Admitted.

End PSA_Pipeline.
